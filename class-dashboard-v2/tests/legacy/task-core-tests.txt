
function taskFixture(){const f=fixture();f.store.dispatch({type:'ADD_STUDENT',seat:1});f.store.dispatch({type:'ADD_STUDENT',seat:2});return f;}
const start=f=>f.store.dispatch({type:'START_TASK',templateId:'reading'});
const check=(f,id,source='STUDENT')=>f.store.dispatch({type:'CHECK_IN',studentId:id,source});
test('有效名單快照、重複打卡拒絕、補登與取消回退',()=>{const f=taskFixture();start(f);check(f,'id-1');assert.throws(()=>check(f,'id-1'),/重複/);check(f,'id-2','TEACHER');f.store.dispatch({type:'UNCHECK',studentId:'id-1'});assert.equal(f.store.getState().rounds[0].checkins.length,1);f.store.dispatch({type:'UNDO_CHECK'});assert.equal(f.store.getState().rounds[0].checkins.length,2);assert.throws(()=>f.store.dispatch({type:'UNDO_CHECK'}));});
test('不參與不計入門檻、進行中名單不隨今日變動',()=>{const f=taskFixture();f.store.dispatch({type:'SET_PARTICIPATION',studentId:'id-2',participating:false});start(f);assert.deepEqual(f.store.getState().rounds[0].participants,['id-1']);assert.equal(f.store.getState().rounds[0].threshold,1);f.store.dispatch({type:'SET_PARTICIPATION',studentId:'id-2',participating:true});assert.throws(()=>check(f,'id-2'),/不在本輪/);});
test('範本快照及歷史保留、單一進行中輪次',()=>{const f=taskFixture();start(f);f.store.dispatch({type:'SAVE_TEMPLATE',id:'reading',name:'新閱讀'});assert.equal(f.store.getState().rounds[0].name,'每日閱讀');assert.throws(()=>start(f));check(f,'id-1');f.store.dispatch({type:'CANCEL_TASK'});assert.throws(()=>check(f,'id-2'));start(f);assert.equal(f.store.getState().rounds.length,2);assert.equal(f.store.getState().rounds[0].checkins.length,1);});
test('任務重載跨日不修改參與名單及進度',()=>{const f=taskFixture();start(f);check(f,'id-1');const reload=createStore({storage:f.storage,now:()=> '2030-01-01T00:00:00Z'});assert.deepEqual(reload.getState().rounds,f.store.getState().rounds);});
test('任務儲存失敗不增加打卡或飼料進度',()=>{const f=taskFixture();start(f);f.storage.setItem=()=>{throw Error('quota')};assert.throws(()=>check(f,'id-1'));assert.equal(f.store.getState().rounds[0].checkins.length,0);});
test('第一階段模型升版保留學生及 ID、讀取不覆寫',()=>{const f=fixture();const raw=JSON.stringify({schemaVersion:1,revision:2,savedAt:null,classroom:{name:'舊班'},students:[{id:'stable',seat:4,name:''}]});f.data.set(STORAGE_KEY,raw);const reload=createStore({storage:f.storage});assert.equal(reload.getState().schemaVersion,3);assert.equal(reload.getState().students[0].id,'stable');assert.equal(f.data.get(STORAGE_KEY),raw);});
