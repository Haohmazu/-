function fixture(){const data=new Map();let id=0;const storage={getItem:k=>data.get(k)??null,setItem:(k,v)=>data.set(k,v)};let day='2026-11-02T01:00:00Z';const store=createStore({storage,newId:()=>`s${++id}`,now:()=>day});return {store,storage,setDay:value=>day=value};}
function setup(){const f=fixture();for(let seat=1;seat<=3;seat++)f.store.dispatch({type:'ADD_STUDENT',seat,name:''});f.store.dispatch({type:'START_TASK',templateId:'reading'});return f;}
const check=(f,id,source='STUDENT')=>f.store.dispatch({type:'CHECK_IN',studentId:id,source});

test('30 位 UTF-8 CSV 匯入',()=>{const f=fixture(),records=rowsToRoster(parseCSV(csvText([['座號','姓名'],...Array.from({length:30},(_,i)=>[i+1,`學生${i+1}`])])));imported(f,records.map(r=>({...r,studentId:null})));assert.equal(f.store.getState().students.length,30);assert.equal(f.store.getState().rosterImports.length,1);});
test('CSV 引號、換行、空姓名與錯誤格式',()=>{assert.deepEqual(parseCSV('座號,姓名\r\n1,"王,同學"\r\n2,"陳""同學"\r\n3,')[1],['1','王,同學']);assert.equal(rowsToRoster(parseCSV('姓名,座號\n,1'))[0].name,'');for(const text of ['座號,姓名\n1,A\n1,B','座號,姓名\n0,A','座號,姓名\n1.2,A','錯欄,姓名\n1,A','座號,姓名\n1,"未結束','座號,姓名\n1,"姓名"錯'])assert.throws(()=>rowsToRoster(parseCSV(text)));assert.ok(csvText([['=HYPERLINK()']]).includes("'=HYPERLINK()"));});
test('錯誤與超額匯入原子拒絕、不覆寫',()=>{const f=setup(),before=f.store.getState();for(const rows of [[{seat:1,name:'a',studentId:null}],[{seat:51,name:'',studentId:null}],[{seat:4,name:'',studentId:'missing'}],Array.from({length:51},(_,i)=>({seat:i+1,name:'',studentId:null}))])assert.throws(()=>imported(f,rows));assert.deepEqual(f.store.getState(),before);});
test('顯式 ID 對應保留學生與歷史、未列入不刪除',()=>{const f=setup();check(f,'s1');const round=f.store.getState().rounds;imported(f,[{seat:1,name:'更新姓名',studentId:'s1'},{seat:4,name:'新學生',studentId:null}]);assert.equal(f.store.getState().students.find(s=>s.seat===1).id,'s1');assert.equal(f.store.getState().students.length,4);assert.deepEqual(f.store.getState().rounds,round);assert.equal(createStore({storage:f.storage}).getState().students[0].name,'更新姓名');});
test('不按姓名推定身分、座號交換顯式對應',()=>{const f=setup();assert.throws(()=>planRoster(f.store.getState(),[{seat:1,name:'',studentId:undefined}]));imported(f,[{seat:2,name:'同名',studentId:'s1'},{seat:1,name:'同名',studentId:'s2'}]);assert.equal(f.store.getState().students.find(s=>s.id==='s1').seat,2);assert.throws(()=>imported(f,[{seat:1,name:'',studentId:'s1'},{seat:2,name:'',studentId:'s1'}]));});
test('預覽後變更與儲存失敗均不匯入',()=>{const f=fixture(),revision=f.store.getState().revision;f.store.dispatch({type:'SET_CLASS_NAME',name:'新班'});assert.throws(()=>f.store.dispatch({type:'IMPORT_STUDENTS',expectedRevision:revision,rows:[{seat:1,name:'',studentId:null}]}),/重新預覽/);f.storage.setItem=()=>{throw Error('full')};assert.throws(()=>imported(f,[{seat:1,name:'',studentId:null}]));assert.equal(f.store.getState().students.length,0);assert.equal(f.store.getState().rosterImports.length,0);});
test('修改參與名單前後紀錄、移除已打卡需要確認',()=>{const f=setup();check(f,'s1');const revision=f.store.getState().revision;assert.throws(()=>f.store.dispatch({type:'REVISE_PARTICIPANTS',participants:['s2','s3'],threshold:2}),/明確確認/);f.store.dispatch({type:'REVISE_PARTICIPANTS',participants:['s2','s3'],threshold:2,confirmRemoveChecked:true,expectedRevision:revision});const r=f.store.getState().rounds[0];assert.equal(r.checkins.length,0);assert.deepEqual(r.audit.at(-1).before.participants,['s1','s2','s3']);assert.equal(r.audit.at(-1).before.checkins.length,1);assert.deepEqual(r.audit.at(-1).after.participants,['s2','s3']);assert.throws(()=>f.store.dispatch({type:'UNDO_CHECK'}));assert.deepEqual(createStore({storage:f.storage}).getState(),f.store.getState());});
test('名單及門檻不合理、重複、未知 ID 拒絕',()=>{const f=setup();for(const [participants,threshold] of [[[],1],[['s1'],2],[['s1'],0],[['s1','s1'],1],[['missing'],1]])assert.throws(()=>f.store.dispatch({type:'REVISE_PARTICIPANTS',participants,threshold}));});
test('補登取消復原統計一致、取消輪次不計有效累計',()=>{const f=setup();check(f,'s1');check(f,'s2','TEACHER');assert.throws(()=>check(f,'s1'));assert.deepEqual(roundStatistics(f.store.getState().rounds[0]),{completed:2,incomplete:1,total:3,rate:2/3});f.store.dispatch({type:'UNCHECK',studentId:'s2'});assert.equal(taskStatistics(f.store.getState()).perStudent.s2,0);f.store.dispatch({type:'UNDO_CHECK'});assert.equal(taskStatistics(f.store.getState()).perStudent.s2,1);assert.throws(()=>f.store.dispatch({type:'UNDO_CHECK'}));f.store.dispatch({type:'CANCEL_TASK'});assert.equal(taskStatistics(f.store.getState()).cancelled,1);assert.equal(taskStatistics(f.store.getState()).perStudent.s1,0);assert.equal(f.store.getState().rounds[0].checkins.length,2);});
test('跨日保留原 ID、名單與打卡，今日參與不影響本輪',()=>{const f=setup();check(f,'s1');const before=f.store.getState().rounds[0];f.setDay('2026-11-03T01:00:00Z');f.store.dispatch({type:'SET_PARTICIPATION',studentId:'s3',participating:false});assert.deepEqual(f.store.getState().rounds[0],before);assert.deepEqual(createStore({storage:f.storage}).getState().rounds[0],before);check(f,'s2');assert.equal(f.store.getState().rounds[0].id,before.id);});
